## ============================================================================
## 记忆碎片模板 - 通用模板
## ============================================================================
## 使用说明：
## 1. 复制此文件并重命名为 memory_XXX_XX.rpy
##    例如：memory_alpha_01.rpy, memory_raven_03.rpy
## 
## 2. 修改下方的参数：
##    - label名称
##    - 角色名（alpha/raven/laurel）
##    - 记忆编号（1-5）
##    - 视角、时间点、主题等信息
## 
## 3. 在"记忆内容"区域填写你的剧情
## 
## 4. 保存到 game/scripts/memories/ 目录
## ============================================================================

## ============================================================================
## 记忆碎片 [编号] - [机体名称]
## ============================================================================
## 视角：[角色名]
## 时间点：[战后第X年]
## 主题：[简短描述这段记忆的内容]
## ============================================================================

label memory_[角色]_0[编号]:
    # 自动标记为已读 - 修改这里的参数
    $ mark_memory_read("[角色]", [编号])
    
    ## ========================================
    ## 记忆开始 - 标题显示
    ## ========================================
    scene black with fade
    
    # 修改颜色代码：
    # 阿尔法用 #0af (蓝色)
    # 鸦羽用 #f90 (橙色)
    # Laurel-II用 #9f0 (绿色)
    centered "{color=#0af}[  记忆碎片  ]{/color}"
    centered "{size=40}[编号]{/size}"
    centered "{size=25}[机体名称] - [视角]{/size}"
    pause 2.0
    
    ## ========================================
    ## 记忆内容 - 在这里填写你的剧情
    ## ========================================
    
    # 设置初始场景
    # scene bg_你的背景 with fade
    
    # 显示角色立绘
    # show 角色名 表情
    
    # 开始写对话
    # "旁白文字"
    # 角色名 "对话内容"
    
    # 示例：
    scene bg_factory with fade
    "这是一个记忆碎片的示例。"
    
    # show lucia normal
    # lucia "在这里写你的对话。"
    
    # show alpha worried
    # alpha "可以有很多对话。"
    # alpha "场景可以切换。"
    
    # scene bg_cliff with dissolve
    # "场景转换到悬崖边..."
    
    "（在这里继续填写你的完整剧情）"
    "（不限长度，想写多少写多少）"
    
    ## ========================================
    ## 可用的Ren'Py命令参考
    ## ========================================
    # scene [背景名] with [转场效果]  - 切换场景
    # show [角色] [表情] at [位置]     - 显示立绘
    # hide [角色]                      - 隐藏立绘
    # 
    # 转场效果：
    # - fade: 淡入淡出
    # - dissolve: 溶解
    # - wipeleft: 从左擦除
    # - pixellate: 像素化
    # 
    # 音效音乐：
    # play music "音乐文件.ogg" fadein 1.0
    # play sound "音效文件.ogg"
    # stop music fadeout 1.0
    # 
    # 暂停和等待：
    # pause 1.5              - 暂停1.5秒
    # with Pause(1.5)        - 带转场的暂停
    
    ## ========================================
    ## 记忆结束
    ## ========================================
    scene black with fade
    centered "{color=#0af}记忆结束{/color}"
    pause 1.0
    
    # 返回工作室
    jump return_to_workshop
